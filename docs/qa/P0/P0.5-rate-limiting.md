// ... other imports

@Module({
  imports: [
    BullModule.forRoot({ ... }),

**СТАЛО:**

```typescript
import { Module } from '@nestjs/common';
import { BullModule } from '@nestjs/bull';
import { ThrottlerModule, ThrottlerGuard } from '@nestjs/throttler';
import { APP_GUARD } from '@nestjs/core';
// ... other imports

@Module({
  imports: [
    // ============ RATE LIMITING ============
    ThrottlerModule.forRoot([
      {
        // Default global rate limit
        name: 'global',
        ttl: 60,           // 60 seconds
        limit: 100,        // 100 requests per 60 seconds
      },
      {
        // Strict limit for authentication endpoints
        name: 'auth',
        ttl: 60,
        limit: 5,          // 5 attempts per 60 seconds (12x per minute limit)
      },
      {
        // Stricter limit for password reset
        name: 'password-reset',
        ttl: 3600,         // 1 hour
        limit: 3,          // 3 attempts per hour
      },
    ]),

    BullModule.forRoot({ ... }),
    PrismaModule,
    RedisModule,
    // ... other imports
  ],
  controllers: [HealthController],
  providers: [
    // ============ REGISTER GLOBAL THROTTLER GUARD ============
    {
      provide: APP_GUARD,
      useClass: ThrottlerGuard,
    },
  ]
})
export class AppModule {}
```

**СТАЛО:**

```typescript
import { Controller, Post, Body } from "@nestjs/common";
import { Throttle } from "@nestjs/throttler";

@Controller("auth")
export class AuthController {
  /**
   * Login endpoint - 5 attempts per minute (strict limit)
   *
   * After 5 failed attempts, requests are rejected for 60 seconds.
   * This prevents brute-force attacks on passwords.
   */
  @Post("login")
  @Throttle("auth") // Uses the 'auth' throttler (5 per 60s)
  async login(@Body() dto: LoginDto) {
    // ...
  }

  /**
   * Register endpoint - also 5 attempts per minute
   *
   * Prevents spam account creation.
   */
  @Post("register")
  @Throttle("auth") // Uses the 'auth' throttler (5 per 60s)
  async register(@Body() dto: RegisterDto) {
    // ...
  }

  /**
   * Password reset - 3 attempts per hour
   *
   * Stricter limit to prevent email bombardment.
   */
  @Post("password-reset")
  @Throttle("password-reset") // Uses the 'password-reset' throttler (3 per 3600s)
  async passwordReset(@Body() dto: PasswordResetDto) {
    // ...
  }

  /**
   * Verify token - use default global throttlar (100 per 60s)
   * Low-security endpoint.
   */
  @Post("verify-token")
  async verifyToken(@Body() dto: VerifyTokenDto) {
    // Uses global throttler (100 per 60s)
    // ...
  }

  /**
   * Logout - no special throttling (user action)
   */
  @Post("logout")
  @Skip() // Skip throttling
  async logout() {
    // ...
  }
}
```

## 📋 ШАГ 4: Создай custom throttle decorator для skip

### 4.1 Якщо потрібен Skip декоратор

**Файл:** `apps/api/src/common/throttle.decorator.ts`

```typescript
import { SetMetadata } from "@nestjs/common";

export const THROTTLE_SKIP_KEY = "THROTTLE_SKIP";

/**
 * Decorator to skip throttling on specific endpoints.
 * Usage: @Throttle.Skip()
 */
export const Throttle = {
  Skip: () => SetMetadata(THROTTLE_SKIP_KEY, true)
};
```

**Потім обнови ThrottlerGuard:**

```typescript
// apps/api/src/app.module.ts (update imports)
import { Reflector } from "@nestjs/core";
import { ThrottlerGuard } from "@nestjs/throttler";
import { THROTTLE_SKIP_KEY } from "@/common/throttle.decorator";

@Injectable()
export class CustomThrottlerGuard extends ThrottlerGuard {
  constructor(private reflector: Reflector) {
    super();
  }

  canActivate(context: ExecutionContext): boolean | Promise<boolean> {
    const skip = this.reflector.getAllAndOverride<boolean>(THROTTLE_SKIP_KEY, [
      context.getHandler(),
      context.getClass()
    ]);

    if (skip) return true;
    return super.canActivate(context);
  }
}

// Змінити APP_GUARD на CustomThrottlerGuard:
providers: [
  {
    provide: APP_GUARD,
    useClass: CustomThrottlerGuard
  }
];
```

## 📋 ШАГ 5: Обнови ответ при rate limit (429)

### 5.1 Создай глобальный exception filter

**Файл:** `apps/api/src/common/throttle-exception.filter.ts`

```typescript
import { ArgumentsHost, Catch, ExceptionFilter, HttpStatus } from "@nestjs/common";
import { ThrottlerException } from "@nestjs/throttler";
ук;
import { Response } from "express";

@Catch(ThrottlerException)
export class ThrottleExceptionFilter implements ExceptionFilter {
  catch(exception: ThrottlerException, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();

    response.status(HttpStatus.TOO_MANY_REQUESTS).json({
      statusCode: HttpStatus.TOO_MANY_REQUESTS,
      message: "Too many requests. Please try again later.",
      retryAfter: response.get("Retry-After") || "60", // Seconds
      timestamp: new Date().toISOString()
    });
  }
}
```

### 5.2 Регистрируй filter в main.ts

```typescript
// apps/api/src/main.ts
import { ThrottleExceptionFilter } from "@/common/throttle-exception.filter";

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  // ... CORS и остальное

  // ============ EXCEPTION FILTERS ============
  app.useGlobalFilters(
    new HttpExceptionFilter(),
    new ThrottleExceptionFilter() // ← Добавь
  );

  // ...
}
```

### 8.1 Коментарі в коді + ADR

**Файл:** `docs/adr/005-rate-limiting.md`

```markdown
# ADR-005: API Rate Limiting

## Problem

Without rate limiting, attackers can:

1. Brute-force passwords (login endpoint)
2. Spam account creation (register)
3. Flood email send (password reset)

## Solution

Use @nestjs/throttler with Redis backend:

### Limits

- **Global**: 100 req/min (default)
- **Auth**: 5 req/min (login, register)
- **Reset**: 3 req/hour (password reset)

### Implementation

- Throttler Guard applied globally
- Per-endpoint @Throttle() decorator override
- @Skip() for non-sensitive endpoints

### Metrics

Track via: `app.get(ThrottlerExpires)` hook

## Monitoring

Alert if auth endpoints receive >50 req/sec (DDoS indicator)
```

## 🚨 ВОЗМОЖНЫЕ ОШИБКИ

| Ошибка                                  | Причина                                          | Решение                                                         |
| --------------------------------------- | ------------------------------------------------ | --------------------------------------------------------------- |
| "Memory store doesn't support metadata" | У-in-memory throttler в development нет всех фич | Используй Redis store для production (конфигурируй `REDIS_URL`) |
| 429 не возвращается после лимита        | Guard не применен глобально                      | Убедись что `APP_GUARD` зарегистрирован в app.module.ts         |
| Throttler не срабатывает                | Endpoint имеет специальное имя?                  | Проверь что все endpoints зарегистрированы в controller         |
| "Key ..."limit exceeded" error в логах  | Redis connection issue                           | Убедись что Redis запущен (или используй in-memory для dev)     |
