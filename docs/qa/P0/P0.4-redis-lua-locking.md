## 🎯 ПРОБЛЕМА

### Текущий код (危險 RACE CONDITION):
```typescript
// apps/api/src/redis/redis.service.ts

async refreshLock(landingId: string, userId: string, ttlSeconds: number = 120): Promise<boolean> {
  const key = `landing:lock:${landingId}`;

  // ⚠️ Step 1: GET (time T1)
  const currentOwner = await this.client.get(key);

  // ⚠️ Step 2: Check in JavaScript (time T1 + δ) - NOT ATOMIC
  if (currentOwner !== userId) {
    return false;
  }

  // ⚠️ Step 3: EXPIRE (time T1 + 2δ)
  // Between Step 1 and Step 3, another user can acquire the lock!
  await this.client.expire(key, ttlSeconds);
  return true;
}
````

### Почему это опасно

```
Timeline:
T0: User A owns lock (key=A, ttl=120)
T1: User A calls refreshLock() → GET returns "A"
T2: User A's lock expires (TTL → 0)
T3: User B calls acquireLock() → SET key B NX EX 120 → SUCCESS (User B owns)
T4: User A's old refreshLock call → EXPIRE key 120 → Extends User B's lock!

Result: User A unknowingly refreshed User B's lock (data corruption)
```

**БЫЛО:**

**СТАЛО:**

```typescript
@Injectable()
export class RedisService {
  private client: redis.RedisClientType;

  // ============ LUA SCRIPTS (ATOMIC OPERATIONS) ============

  /**
   * Atomic compare-and-delete script.
   * Returns 1 if lock was owned by userId and deleted, 0 otherwise.
   *
   * KEYS[1] = lock key
   * ARGV[1] = expected userId
   *
   * Redis Lua scripts are atomic - either all commands execute or none.
   */
  private readonly releaseLockScript = `
    if redis.call('get', KEYS[1]) == ARGV[1] then
      return redis.call('del', KEYS[1])
    else
      return 0
    end
  `;

  /**
   * Atomic compare-and-extend script.
   * Returns 1 if lock was extended, 0 otherwise.
   *
   * KEYS[1] = lock key
   * ARGV[1] = expected userId
   * ARGV[2] = new TTL in seconds
   */
  private readonly refreshLockScript = `
    if redis.call('get', KEYS[1]) == ARGV[1] then
      return redis.call('expire', KEYS[1], ARGV[2])
    else
      return 0
    end
  `;

  /**
   * Atomic compare-and-set script (for lock acquisition with verification).
   * Ensures only one user can own the lock at a time.
   *
   * KEYS[1] = lock key
   * ARGV[1] = userId (new owner)
   * ARGV[2] = TTL in seconds
   */
  private readonly acquireLockScript = `
    if redis.call('get', KEYS[1]) == false then
      return redis.call('set', KEYS[1], ARGV[1], 'EX', ARGV[2], 'NX')
    else
      return nil
    end
  `;

  // ============ PUBLIC METHODS (ATOMIC) ============

  /**
   * Atomically release a lock if owned by userId.
   */
  async releaseLock(key: string, userId: string): Promise<boolean> {
    try {
      // eval(script, numKeys, key1, key2, ..., arg1, arg2, ...)
      const result = await this.client.eval(this.releaseLockScript, {
        keys: [key],
        arguments: [userId]
      });

      // Lua returns 1 if deleted, 0 if not owner
      return result === 1;
    } catch (error) {
      console.error(`[RedisService] releaseLock failed for key=${key}:`, error);
      throw error;
    }
  }

  /**
   * Atomically extend lock TTL if owned by userId.
   */
  async refreshLock(
    key: string,
    userId: string,
    ttlSeconds: number = 120
  ): Promise<boolean> {
    try {
      const result = await this.client.eval(this.refreshLockScript, {
        keys: [key],
        arguments: [userId, ttlSeconds.toString()]
      });

      // Lua returns 1 if TTL was extended, 0 if not owner
      return result === 1;
    } catch (error) {
      console.error(`[RedisService] refreshLock failed for key=${key}:`, error);
      throw error;
    }
  }

  /**
   * Atomically acquire a lock.
   * Returns the owner userId if successful, null if already owned.
   */
  async acquireLock(
    key: string,
    userId: string,
    ttlSeconds: number = 120
  ): Promise<boolean> {
    try {
      // SET with NX (only if not exists) is atomic in Redis
      const result = await this.client.set(key, userId, {
        EX: ttlSeconds,
        NX: true // Only set if key doesn't exist
      });

      // result is 'OK' if set, null if already exists
      return result === "OK";
    } catch (error) {
      console.error(`[RedisService] acquireLock failed for key=${key}:`, error);
      throw error;
    }
  }

  /**
   * Get current lock owner.
   */
  async getLockOwner(key: string): Promise<string | null> {
    try {
      return await this.client.get(key);
    } catch (error) {
      console.error(`[RedisService] getLockOwner failed for key=${key}:`, error);
      return null;
    }
  }
}
```

```typescript
async lock(landingId: string, userId: string) {
  const key = `landing:lock:${landingId}`;

  // ============ ATTEMPT 1: Try to acquire lock ============
  const lockAcquired = await this.redis.acquireLock(key, userId, 120);

  if (lockAcquired) {
    // ✅ Successfully acquired - write to Postgres
    try {
      await this.prisma.landing.update({
        where: { id: landingId },
        data: {
          lockedById: userId,
          lockedAt: new Date(),
          lockExpires: new Date(Date.now() + 120 * 1000),
        },
      });

      // ✅ Audit log
      await this.auditService.log({
        action: 'LANDING_LOCKED',
        landingId,
        userId,
        details: { acquiredNewLock: true },
      });

      return {
        locked: true,
        owner: userId,
        isNewAcquisition: true,
      };
    } catch (dbError) {
      // 🚨 If Postgres write fails, release the Redis lock
      await this.redis.releaseLock(key, userId);
      throw dbError;
    }
  }

  // ============ ATTEMPT 2: Check if current owner ============
  const currentOwner = await this.redis.getLockOwner(key);

  if (currentOwner === userId) {
    // ✅ Re-entrant lock - refresh TTL (ATOMIC via Lua)
    const refreshed = await this.redis.refreshLock(key, userId, 120);

    if (refreshed) {
      // ✅ TTL extended successfully

      // Update Postgres lock timestamp
      await this.prisma.landing.update({
        where: { id: landingId },
        data: {
          lockExpires: new Date(Date.now() + 120 * 1000),
        },
      });

      await this.auditService.log({
        action: 'LANDING_LOCK_REFRESHED',
        landingId,
        userId,
      });

      return {
        locked: true,
        owner: userId,
        isNewAcquisition: false,
        refreshed: true,
      };
    } else {
      // 🚨 Race condition: We thought we owned it, but refresh failed
      // (lock may have expired and been acquired by someone else)
      throw new ConflictException(
        'Lock ownership changed during refresh. Please try again.'
      );
    }
  } else {
    // ❌ Another user owns the lock
    throw new ConflictException(
      `Landing is locked by user ${currentOwner || 'unknown'}. Try again later.`
    );
  }
}
```

### 3.3 Обнови метод unlock()

```typescript
async unlock(landingId: string, userId: string) {
  const key = `landing:lock:${landingId}`;

  // ✅ Atomic release (only if owned by userId)
  const released = await this.redis.releaseLock(key, userId);

  if (!released) {
    // Not owner or already released
    console.warn(
      `[LandingService] Unlock failed - not owner. landingId=${landingId}, userId=${userId}`
    );
    // Still clear Postgres record (best effort)
  }

  // Clear Postgres lock record
  await this.prisma.landing.update({
    where: { id: landingId },
    data: {
      lockedById: null,
      lockedAt: null,
      lockExpires: null,
    },
  });

  await this.auditService.log({
    action: 'LANDING_UNLOCKED',
    landingId,
    userId,
  });

  return { unlocked: true };
}

import { Test, TestingModule } from "@nestjs/testing";
import { RedisService } from "./redis.service";

describe("RedisService - Atomic Locks", () => {
  let service: RedisService;

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [RedisService]
    }).compile();

    service = module.get<RedisService>(RedisService);
  });

  describe("Lua atomicity", () => {
    it("should atomically release lock only if owner matches", async () => {
      const key = "test:lock:123";
      const userA = "user-a";
      const userB = "user-b";

      // User A acquires lock
      await service.acquireLock(key, userA, 60);

      // User B tries to release (should fail)
      const releasedByB = await service.releaseLock(key, userB);
      expect(releasedByB).toBe(false);

      // Verify lock still exists (owned by A)
      const owner = await service.getLockOwner(key);
      expect(owner).toBe(userA);

      // User A releases (should succeed)
      const releasedByA = await service.releaseLock(key, userA);
      expect(releasedByA).toBe(true);

      // Lock should be gone
      const ownerAfterRelease = await service.getLockOwner(key);
      expect(ownerAfterRelease).toBeNull();
    });

    it("should atomically refresh lock TTL only if owner matches", async () => {
      const key = "test:lock:456";
      const userA = "user-a";
      const userB = "user-b";

      // User A acquires lock
      await service.acquireLock(key, userA, 10);

      // User B tries to refresh (should fail)
      const refreshedByB = await service.refreshLock(key, userB, 60);
      expect(refreshedByB).toBe(false);

      // User A refreshes (should succeed)
      const refreshedByA = await service.refreshLock(key, userA, 60);
      expect(refreshedByA).toBe(true);

      // Verify still owned by A
      const owner = await service.getLockOwner(key);
      expect(owner).toBe(userA);
    });
  });
});

# ADR-004: Redis Atomic Locking via Lua Scripts

## Problem

GET → check → SET/EXPIRE pattern is not atomic in Redis.
Between GET and SET, another client can modify the key, leading to race conditions.

## Solution

Use Redis Lua scripts for atomic operations:

- Lua scripts execute atomically on Redis server
- Either all commands in script execute or none
- No time window for race conditions

## Implementation

Three atomic operations:

1. **acquireLock**: SET key value EX ttl NX (atomic in Redis 3.2+)
2. **releaseLock**: Lua script - GET, compare, DEL (atomic)
3. **refreshLock**: Lua script - GET, compare, EXPIRE (atomic)

## Testing

- Unit tests in redis.service.spec.ts
- E2E: Concurrent lock attempts
- Load test: 100+ concurrent users on same landing

## Security

- Lock ownership verified before modification
- Lua scripts execute on server (no network race window)
- Short TTL prevents zombie locks
```
