## 📋 ШАГ 1: Создай Zod Schema для GrapesJSON

### 1.1 Створи файл

**Файл:** `packages/types/src/grapes-document.schema.ts`

```typescript
import { z } from "zod";

/**
 * Zod schema for GrapesJS document structure.
 *
 * This ensures type safety when storing/loading editor documents.
 *
 * See: https://grapesjs.com/guide/components/component.html
 */

// ============ ASSET SCHEMA ============
export const GrapesAssetSchema = z
  .object({
    /** Asset ID */
    id: z.string().optional(),

    /** Asset URL */
    src: z.string().url(),

    /** Asset name */
    name: z.string().optional(),

    /** Asset type: image, video, etc */
    type: z.enum(["image", "video", "font"]).optional(),

    /** Additional metadata */
    media: z.string().optional()

    // Allow unknown properties (GrapesJS may add more)
  })
  .passthrough();

export type GrapesAsset = z.infer<typeof GrapesAssetSchema>;

// ============ COMPONENT SCHEMA ============
export const GrapesComponentSchema = z
  .object({
    /** Component ID (unique) */
    tagName: z.string().default("div"),

    /** Component content */
    content: z.string().optional(),

    /** Component attributes */
    attributes: z.record(z.any()).optional(),

    /** Component styles */
    style: z.record(z.number().or(z.string())).optional(),

    /** Child components */
    components: z.lazy(() => GrapesComponentSchema.array()).optional(),

    /** Component traits (properties) */
    traits: z.array(z.record(z.any())).optional()

    // Allow unknown properties
  })
  .passthrough();

export type GrapesComponent = z.infer<typeof GrapesComponentSchema>;

// ============ STYLE RULE SCHEMA ============
export const GrapesStyleSchema = z
  .object({
    /** CSS selector */
    selectors: z.string().array(),

    /** CSS property: value pairs */
    properties: z.record(z.string()).optional()

    // Allow unknown properties
  })
  .passthrough();

export type GrapesStyle = z.infer<typeof GrapesStyleSchema>;

// ============ MAIN GRAPES DOCUMENT SCHEMA ============
export const GrapesDocumentSchema = z
  .object({
    /** Asset list */
    assets: z.array(GrapesAssetSchema).optional().default([]),

    /** Component tree */
    components: z.array(GrapesComponentSchema).optional().default([]),

    /** Style rules */
    styles: z.array(GrapesStyleSchema).optional().default([]),

    /** Rendered HTML (derived from components) */
    html: z.string().optional().default(""),

    /** Rendered CSS (derived from styles) */
    css: z.string().optional().default("")

    // Allow unknown properties for future extensions
  })
  .passthrough();

export type GrapesDocument = z.infer<typeof GrapesDocumentSchema>;

/**
 * Safely parse GrapesJSON from database.
 *
 * Returns parsed document or throws ZodError.
 *
 * Usage:
 * const doc = parseGrapesDocument(v.grapesJson);
 */
export function parseGrapesDocument(data: unknown): GrapesDocument {
  return GrapesDocumentSchema.parse(data);
}

/**
 * Validate and fix GrapesJSON with sensible defaults.
 *
 * Returns parsed document or throws error with details.
 * Useful for migrations/imports.
 */
export function validateGrapesDocument(data: unknown): {
  success: boolean;
  data?: GrapesDocument;
  errors?: z.ZodError;
} {
  const result = GrapesDocumentSchema.safeParse(data);
  if (result.success) {
    return { success: true, data: result.data };
  } else {
    return { success: false, errors: result.error };
  }
}
```

```typescript
import { parseGrapesDocument, GrapesDocument } from "@workspace/types";

@Injectable()
export class LandingsService {
  // ...

  async editor(landingId: string): Promise<GrapesDocument> {
    const landing = await this.prisma.landing.findUnique({
      where: { id: landingId },
      include: { currentVersion: true }
    });

    if (!landing || !landing.currentVersion) {
      throw new NotFoundException("Landing or version not found");
    }

    const v = landing.currentVersion;

    try {
      // ✅ Parse with validation
      const grapesJson = parseGrapesDocument(v.grapesJson);

      return grapesJson;
    } catch (error) {
      if (error instanceof z.ZodError) {
        // Log detailed error for debugging
        console.error("[LandingsService] Invalid GrapesJSON:", {
          landingId,
          versionId: v.id,
          errors: error.errors
        });

        // Return empty but valid document
        // (Better than throwing 500 error to the user)
        return parseGrapesDocument({
          assets: [],
          components: [],
          styles: [],
          html: "<div>Failed to load editor document</div>",
          css: ""
        });
      }

      throw error;
    }
  }
}
```

### 3.1 Обновить useEditorDocument

**Файл:** `apps/web/hooks/editor/use-editor-document.ts`

```typescript
import { useQuery } from "@tanstack/react-query";
import { GrapesDocument, parseGrapesDocument } from "@workspace/types";
import { apiClient } from "@/lib/api-client";

export interface UseEditorDocumentOptions {
  landingId: string;
  enabled?: boolean;
}

/**
 * Hook to fetch and validate editor document.
 *
 * Returns parsed GrapesDocument with full type safety.
 */
export function useEditorDocument({
  landingId,
  enabled = true
}: UseEditorDocumentOptions) {
  return useQuery({
    queryKey: ["landing", landingId, "editor"],
    queryFn: async () => {
      const response = await apiClient.get(`/landings/${landingId}/editor`);

      // ✅ Validate on client side too
      const doc = parseGrapesDocument(response.data);

      return doc;
    },
    staleTime: 1000 * 60,
    gcTime: 1000 * 60 * 5,
    enabled,
    retry: (failureCount) => {
      // Don't retry on validation errors (4xx)
      return failureCount < 1;
    }
  });
}
```

### 3.2 Обновить useGrapesSerialize

```typescript
import { GrapesDocument, GrapesDocumentSchema } from "@workspace/types";
import { Editor } from "grapesjs";

export function useGrapesSerialize() {
  const grapeEditorRef = useRef<Editor | null>(null);

  const serialize = useCallback((): GrapesDocument => {
    if (!grapeEditorRef.current) {
      console.warn("[useGrapesSerialize] Editor not initialized");
      return GrapesDocumentSchema.parse({});
    }

    // Serialize Grapes document
    const projectData = grapeEditorRef.current.getProjectData();

    // ✅ Validate before returning
    const validated = GrapesDocumentSchema.parse({
      assets: projectData.assets || [],
      components: projectData.components || [],
      styles: projectData.styles || [],
      html: grapeEditorRef.current.getHtml(),
      css: grapeEditorRef.current.getCss()
    });

    return validated;
  }, []);

  return { setGrapeEditor, serialize };
}
```

## 📋 ШАГ 4: Migration: Fix존하는 bad GrapesJSON

### 4.1 Создай migration script

**Файл:** `apps/api/src/migrations/fix-grapes-json.ts`

```typescript
import { PrismaClient } from "@prisma/client";
import { parseGrapesDocument, validateGrapesDocument } from "@workspace/types";

/**
 * Migration to validate and fix existing GrapesJSON documents.
 *
 * Run: npx ts-node src/migrations/fix-grapes-json.ts
 */

const prisma = new PrismaClient();

async function main() {
  console.log("🔍 Scanning for invalid GrapesJSON documents...");

  const versions = await prisma.version.findMany({
    where: {
      grapesJson: {
        not: null
      }
    }
  });

  console.log(`Found ${versions.length} versions to check`);

  let fixed = 0;
  let errors = 0;

  for (const version of versions) {
    const result = validateGrapesDocument(version.grapesJson);

    if (!result.success) {
      console.warn(
        `❌ Version ${version.id} has invalid GrapesJSON:`,
        result.errors?.errors
      );

      try {
        // Try to fix with defaults
        const fixed = parseGrapesDocument({
          assets: Array.isArray(version.grapesJson?.assets)
            ? version.grapesJson.assets
            : [],
          components: Array.isArray(version.grapesJson?.components)
            ? version.grapesJson.components
            : [],
          styles: Array.isArray(version.grapesJson?.styles)
            ? version.grapesJson.styles
            : [],
          html:
            typeof version.grapesJson?.html === "string" ? version.grapesJson.html : "",
          css: typeof version.grapesJson?.css === "string" ? version.grapesJson.css : ""
        });

        await prisma.version.update({
          where: { id: version.id },
          data: { grapesJson: fixed as any }
        });

        console.log(`✅ Fixed version ${version.id}`);
        fixed++;
      } catch (err) {
        console.error(`❌ Could not fix version ${version.id}:`, err);
        errors++;
      }
    }
  }

  console.log(`\n📊 Summary:`);
  console.log(`  Fixed: ${fixed}`);
  console.log(`  Errors: ${errors}`);
  console.log(`  OK: ${versions.length - fixed - errors}`);
}

main()
  .catch(console.error)
  .finally(() => prisma.$disconnect());
```
