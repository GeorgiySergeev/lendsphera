**Время:** 2 часа

## 🎯 ПРОБЛЕМА

### Текущее состояние (危險 PERFORMANCE):
```typescript
// Симуляция текущей реализации
useEffect(() => {
  const interval = setInterval(() => {
    // Every 10 seconds на main thread:
    const serialized = JSON.stringify(editor.serialize()); // 5-50 MB stringify!

    // Compare entire string
    if (serialized !== lastSnapshot) {
      // Send
    }
  }, 10000);
}, []);

// Проблема:
// 1. JSON.stringify на 5-50 MB редактора → blocking main thread
// 2. Браузер фризит на 100-500ms на complex landings
// 3. Happens on EVERY 10s, regardless of actual changes
````

### Оптимальная схема

```typescript
// Instead:
// 1. GrapesJS emits events: component:update, style:update, etc
// 2. On ANY event → set dirtyFlag = true (instant)
// 3. Autosave interval checks: "is dirtyFlag true?" (O(1), не serialize)
// 4. Only if dirty → serialize once and send
```

---

## 📋 АРХИТЕКТУРА ПОСЛЕ РЕФАКТОРИНГА

```
GrapesJS Editor
  ├─ on('component:update') → markDirty() [instant]
  ├─ on('style:update') → markDirty() [instant]
  ├─ on('asset:*') → markDirty() [instant]
  │
  └─ Autosave Interval (every 10s)
     ├─ Check: isDirty? [O(1)]
     ├─ If NO → skip
     └─ If YES:
        ├─ serialize() once [done once per save]
        ├─ send to API
        ├─ set isDirty = false
```

**Impact:**

- Before: 10s × JSON.stringify = ~40 serialize/minute = ~50MB transfers/minute
- After: Only serialize when actually changed (maybe 5-10× per minute during active editing)

### 1.2 Обнови GrapesCanvas компонент

**Файл:** `apps/web/components/editor/grapes-canvas.tsx`

```typescript
"use client";

import React, { useRef, useEffect, useCallback } from 'react';
import grapesjs, { Editor } from 'grapesjs';
import { useEditor } from '@/context/editor-context';

export interface GrapesCanvasProps {
  landing: any;
}

/**
 * GrapesJS Canvas initialized with dirty flag tracking.
 *
 * Emits events on any change, which triggers autosave dirty flag.
 */
export const GrapesCanvas = React.forwardRef<Editor | null, GrapesCanvasProps>(
  function GrapesCanvas({ landing }, ref) {
    const editor = useEditor();
    const containerRef = useRef<HTMLDivElement>(null);
    const localEditorRef = useRef<Editor | null>(null);

    // ============ INITIALIZE GRAPES EDITOR ============
    useEffect(() => {
      if (!containerRef.current || !landing) return;

      // Create GrapesJS editor
      const grapesEditor = grapesjs.init({
        container: containerRef.current,
        components: landing.grapesJson?.components || [],
        style: landing.grapesJson?.styles || [],
        html: landing.html || '<div>Start editing</div>',
        css: landing.css || '',
        storageManager: false, // We handle storage via API
        noticeOnUnload: false,
        // ... other GrapesJS config
      });

      localEditorRef.current = grapesEditor;

      // Forward ref
      if (typeof ref === 'function') {
        ref(grapesEditor);
      } else if (ref) {
        ref.current = grapesEditor;
      }

      // ============ WIRE UP DIRTY FLAG ON GRAPES EVENTS ============
      const markDirtyCallback = () => {
        console.debug('[GrapesCanvas] Change detected - marking dirty');
        editor.autosave.markDirty();
      };

      // Component changes
      grapesEditor.on('component:add', markDirtyCallback);
      grapesEditor.on('component:remove', markDirtyCallback);
      grapesEditor.on('component:update', markDirtyCallback);
      grapesEditor.on('component:mount', markDirtyCallback);
      grapesEditor.on('component:unmount', markDirtyCallback);

      // Style changes
      grapesEditor.on('style:add', markDirtyCallback);
      grapesEditor.on('style:remove', markDirtyCallback);
      grapesEditor.on('style:update', markDirtyCallback);

      // Asset changes
      grapesEditor.on('asset:add', markDirtyCallback);
      grapesEditor.on('asset:remove', markDirtyCallback);
      grapesEditor.on('asset:update', markDirtyCallback);

      // Selection changes (NOT dirty, just for UI)
      grapesEditor.on('component:selected', (component) => {
        editor.ui.selectWidget(component.id);
      });

      grapesEditor.on('component:deselected', () => {
        editor.ui.selectWidget(null);
      });

      // Undo/Redo
      grapesEditor.on('undo', markDirtyCallback);
      grapesEditor.on('redo', markDirtyCallback);

      // Cleanup
      return () => {
        // Unregister all listeners
        grapesEditor.off('component:add', markDirtyCallback);
        grapesEditor.off('component:remove', markDirtyCallback);
        grapesEditor.off('component:update', markDirtyCallback);
        grapesEditor.off('style:add', markDirtyCallback);
        grapesEditor.off('style:remove', markDirtyCallback);
        grapesEditor.off('style:update', markDirtyCallback);
        grapesEditor.off('asset:add', markDirtyCallback);
        grapesEditor.off('asset:remove', markDirtyCallback);
        grapesEditor.off('undo', markDirtyCallback);
        grapesEditor.off('redo', markDirtyCallback);

        grapesEditor.destroy();
        localEditorRef.current = null;
      };
    }, [landing, editor.autosave, editor.ui]);

    // ============ EXPOSE SERIALIZE FUNCTION ============
    // The autosave hook needs a way to serialize the document
    useEffect(() => {
      if (!localEditorRef.current) return;

      // Store serialize function in a way autosave hook can access it
      // (This is done via a ref or callback - see below)
    }, []);

    return (
      <div
        ref={containerRef}
        id="grapes-editor"
        className="flex-1 bg-white overflow-hidden"
        style={{ minHeight: '500px' }}
      >
        {/* GrapesJS renders here */}
      </div>
    );
  }
);

GrapesCanvas.displayName = 'GrapesCanvas';
```

## 📋 ШАГ 2: Создай useGrapeSerialize hook

### 2.1 Hook для управления сериализацией

**Файл:** `apps/web/hooks/editor/use-grapes-serialize.ts`

```typescript
import { useRef, useCallback } from "react";
import { Editor } from "grapesjs";

/**
 * Hook to expose a serialize function for autosave.
 *
 * This separates the serialization logic from the editor initialization,
 * so autosave can call it on-demand (only when dirty).
 */
export function useGrapesSerialize() {
  const grapeEditorRef = useRef<Editor | null>(null);

  const setGrapeEditor = useCallback((editor: Editor | null) => {
    grapeEditorRef.current = editor;
  }, []);

  const serialize = useCallback(() => {
    if (!grapeEditorRef.current) {
      console.warn("[useGrapesSerialize] Editor not initialized");
      return {};
    }

    // Serialize the entire Grapes document
    const serialized = grapeEditorRef.current.getProjectData();

    return {
      assets: serialized.assets,
      components: serialized.components,
      styles: serialized.styles,
      html: grapeEditorRef.current.getHtml(),
      css: grapeEditorRef.current.getCss()
      // Don't include the full editor config, just the data
    };
  }, []);

  return {
    setGrapeEditor,
    serialize
  };
}
```

## 📋 ШАГ 3: Обновить EditorProvider для wire-up serialization

### 3.1 Обнови EditorProvider

**Файл:** `apps/web/context/editor-context.tsx` (обновить)

```typescript
"use client";

import React, { createContext, ReactNode, useMemo, useCallback } from 'react';
import { useEditorLock } from '@/hooks/editor';
import { useEditorAutosave } from '@/hooks/editor';
import { useEditorDocument } from '@/hooks/editor';
import { useEditorKeyboard } from '@/hooks/editor';
import { useEditorCustomCss } from '@/hooks/editor';
import { useEditorUi } from '@/hooks/editor';
import { useGrapesSerialize } from '@/hooks/editor/use-grapes-serialize';
import { Editor } from 'grapesjs';

export interface EditorContextValue {
  lock: ReturnType<typeof useEditorLock>;
  autosave: ReturnType<typeof useEditorAutosave>;
  document: ReturnType<typeof useEditorDocument>;
  ui: ReturnType<typeof useEditorUi>;
  customCss: ReturnType<typeof useEditorCustomCss>;

  // ✅ NEW: callbacks to setup Grapes integration
  setGrapeEditor: (editor: Editor | null) => void;
}

export const EditorContext = createContext<EditorContextValue | null>(null);

export interface EditorProviderProps {
  landingId: string;
  userId: string;
  children: ReactNode;
}

export function EditorProvider({
  landingId,
  userId,
  children,
}: EditorProviderProps) {
  const lock = useEditorLock({ landingId, userId });

  const { setGrapeEditor, serialize } = useGrapesSerialize();

  // ============ AUTOSAVE WITH SERIALIZE ============
  const autosave = useEditorAutosave({
    landingId,
    serializeEditor: serialize, // ✅ Pass the serialize function!
  });

  const document = useEditorDocument({ landingId });
  const ui = useEditorUi();
  const customCss = useEditorCustomCss();

  useEditorKeyboard({
    handlers: {
      onSave: () => {
        console.log('Save shortcut triggered');
      },
    },
  });

  const value: EditorContextValue = useMemo(
    () => ({
      lock,
      autosave,
      document,
      ui,
      customCss,
      setGrapeEditor, // ✅ Expose this so Canvas can call it
    }),
    [lock, autosave, document, ui, customCss, setGrapeEditor]
  );

  return (
    <EditorContext.Provider value={value}>
      {children}
    </EditorContext.Provider>
  );
}

export function useEditor() {
  const context = React.useContext(EditorContext);
  if (!context) {
    throw new Error('useEditor must be used within EditorProvider');
  }
  return context;
}
```
