```typescript
import { useEffect, useRef, useState, useCallback } from "react";
import { useQuery, useMutation } from "@tanstack/react-query";
import { apiClient } from "@/lib/api-client";

export type LockStatus = "unlocked" | "acquiring" | "acquired" | "error";

export interface EditorLockState {
  status: LockStatus;
  ownerId: string | null;
  acquiredAt: Date | null;
  expiresAt: Date | null;
}

export interface UseEditorLockOptions {
  landingId: string;
  userId: string;
  ttl?: number; // heartbeat interval in seconds
  enabled?: boolean;
}

/**
 * Hook to manage editor lock acquisition and heartbeat.
 *
 * Handles:
 * - Initial lock acquisition
 * - Automatic heartbeat to keep lock alive
 * - Lock release on unmount
 * - Conflict handling (another user locked)
 */
export function useEditorLock({
  landingId,
  userId,
  ttl = 30,
  enabled = true
}: UseEditorLockOptions) {
  const [lockState, setLockState] = useState<EditorLockState>({
    status: "unlocked",
    ownerId: null,
    acquiredAt: null,
    expiresAt: null
  });

  const heartbeatIntervalRef = useRef<NodeJS.Timeout | null>(null);

  // ============ ACQUIRE LOCK ============
  const { mutate: acquireLock } = useMutation({
    mutationFn: async () => {
      const response = await apiClient.post(`/landings/${landingId}/lock`, {
        userId
      });
      return response.data;
    },
    onSuccess: (data) => {
      setLockState({
        status: "acquired",
        ownerId: data.owner,
        acquiredAt: new Date(),
        expiresAt: new Date(Date.now() + ttl * 1000)
      });
    },
    onError: (error: any) => {
      console.error("[useEditorLock] Acquire failed:", error);
      setLockState((prev) => ({
        ...prev,
        status: "error"
      }));
    }
  });

  // ============ REFRESH LOCK (HEARTBEAT) ============
  const { mutate: refreshLock } = useMutation({
    mutationFn: async () => {
      const response = await apiClient.post(`/landings/${landingId}/lock/refresh`, {
        userId
      });
      return response.data;
    },
    onSuccess: () => {
      setLockState((prev) => ({
        ...prev,
        expiresAt: new Date(Date.now() + ttl * 1000)
      }));
    },
    onError: (error: any) => {
      console.error("[useEditorLock] Refresh failed:", error);
      // On refresh fail, lock is probably expired
      setLockState((prev) => ({
        ...prev,
        status: "error"
      }));
    }
  });

  // ============ RELEASE LOCK ============
  const { mutate: releaseLock } = useMutation({
    mutationFn: async () => {
      await apiClient.post(`/landings/${landingId}/lock/release`, {
        userId
      });
    },
    onSuccess: () => {
      setLockState({
        status: "unlocked",
        ownerId: null,
        acquiredAt: null,
        expiresAt: null
      });
    }
  });

  // ============ INITIAL LOCK ACQUISITION ============
  useEffect(() => {
    if (!enabled) return;

    setLockState((prev) => ({ ...prev, status: "acquiring" }));
    acquireLock();
  }, [enabled, landingId, userId]);

  // ============ HEARTBEAT TO KEEP LOCK ALIVE ============
  useEffect(() => {
    if (lockState.status !== "acquired") return;

    // First heartbeat after 2/3 of TTL (so we have time to retry if needed)
    const heartbeatInterval = ((ttl * 2) / 3) * 1000;

    heartbeatIntervalRef.current = setInterval(() => {
      console.debug("[useEditorLock] Heartbeat...");
      refreshLock();
    }, heartbeatInterval);

    return () => {
      if (heartbeatIntervalRef.current) {
        clearInterval(heartbeatIntervalRef.current);
      }
    };
  }, [lockState.status, ttl, userId, refreshLock]);

  // ============ CLEANUP: RELEASE LOCK ON UNMOUNT ============
  useEffect(() => {
    return () => {
      if (lockState.status === "acquired") {
        console.debug("[useEditorLock] Unmounting - releasing lock");
        releaseLock();
      }
    };
  }, [lockState.status, releaseLock]);

  return lockState;
}
```

```typescript
import { useEffect, useRef, useState, useCallback, useMemo } from "react";
import { useMutation } from "@tanstack/react-query";
import { apiClient } from "@/lib/api-client";

export interface UseEditorAutosaveOptions {
  landingId: string;
  serializeEditor: () => Record<string, any>; // Function to serialize GrapesJS doc
  intervalMs?: number; // Check every N ms (default 10000 = 10s)
  enabled?: boolean;
}

export interface AutosaveState {
  status: "idle" | "saving" | "saved" | "error";
  isDirty: boolean;
  lastSavedAt: Date | null;
  error: Error | null;
}

/**
 * Hook to manage editor autosave.
 *
 * Uses a dirty flag instead of stringifying entire document every 10s.
 *
 * Handles:
 * - Track dirty flag from GrapesJS events
 * - Serialize and send only when dirty
 * - Handle save errors gracefully
 * - Track save status and timestamps
 */
export function useEditorAutosave({
  landingId,
  serializeEditor,
  intervalMs = 10000,
  enabled = true
}: UseEditorAutosaveOptions) {
  const [autosaveState, setAutosaveState] = useState<AutosaveState>({
    status: "idle",
    isDirty: false,
    lastSavedAt: null,
    error: null
  });

  const dirtyFlagRef = useRef(false);
  const autosaveIntervalRef = useRef<NodeJS.Timeout | null>(null);

  // ============ SAVE MUTATION ============
  const { mutate: saveMutation } = useMutation({
    mutationFn: async (payload: Record<string, any>) => {
      const response = await apiClient.patch(`/landings/${landingId}`, {
        grapesJson: payload
      });
      return response.data;
    },
    onSuccess: () => {
      setAutosaveState((prev) => ({
        ...prev,
        status: "saved",
        isDirty: false,
        lastSavedAt: new Date(),
        error: null
      }));

      // Reset to idle after 2 seconds
      setTimeout(() => {
        setAutosaveState((prev) => ({
          ...prev,
          status: prev.isDirty ? "saving" : "idle"
        }));
      }, 2000);
    },
    onError: (error: Error) => {
      console.error("[useEditorAutosave] Save failed:", error);
      setAutosaveState((prev) => ({
        ...prev,
        status: "error",
        error
      }));
    }
  });

  // ============ SET DIRTY FLAG (called from GrapesJS events) ============
  const markDirty = useCallback(() => {
    dirtyFlagRef.current = true;
    setAutosaveState((prev) => ({ ...prev, isDirty: true }));
  }, []);

  // ============ AUTOSAVE INTERVAL ============
  useEffect(() => {
    if (!enabled) return;

    autosaveIntervalRef.current = setInterval(() => {
      // Only save if dirty
      if (dirtyFlagRef.current) {
        console.debug("[useEditorAutosave] Autosaving...");
        setAutosaveState((prev) => ({ ...prev, status: "saving" }));

        // Serialize the editor document
        const payload = serializeEditor();

        // Send to API
        saveMutation(payload);

        // Clear dirty flag (will be set again if more changes happen)
        dirtyFlagRef.current = false;
      }
    }, intervalMs);

    return () => {
      if (autosaveIntervalRef.current) {
        clearInterval(autosaveIntervalRef.current);
      }
    };
  }, [enabled, intervalMs, serializeEditor, saveMutation]);

  return {
    ...autosaveState,
    markDirty
  };
}
```

```typescript
import { useQuery } from "@tanstack/react-query";

export interface GrapesDocument {
  id: string;
  name: string;
  html: string;
  css: string;
  grapesJson: Record<string, any>;
  customCss: string;
  placeholders: Record<string, any>;
}

export interface UseEditorDocumentOptions {
  landingId: string;
  enabled?: boolean;
}

/**
 * Hook to fetch and cache editor document.
 *
 * Handles:
 * - Fetch landing data by ID
 * - Parse GrapesJSON schema
 * - Memoize document to prevent unnecessary re-renders
 * - Error handling
 */
export function useEditorDocument({
  landingId,
  enabled = true
}: UseEditorDocumentOptions) {
  return useQuery({
    queryKey: ["landing", landingId, "editor"],
    queryFn: async () => {
      const response = await apiClient.get(`/landings/${landingId}/editor`);
      return response.data as GrapesDocument;
    },
    staleTime: 1000 * 60, // 1 minute
    gcTime: 1000 * 60 * 5, // 5 minutes
    enabled
  });
}
```

```typescript
import { useEffect, useCallback } from "react";

export interface KeyboardHandlers {
  onSave: () => void;
  onUndo?: () => void;
  onRedo?: () => void;
  onDelete?: () => void;
}

export interface UseEditorKeyboardOptions {
  handlers: KeyboardHandlers;
  enabled?: boolean;
}

/**
 * Hook to handle keyboard shortcuts in editor.
 *
 * Shortcuts:
 * - Ctrl+S / Cmd+S: Save
 * - Ctrl+Z / Cmd+Z: Undo
 * - Ctrl+Shift+Z / Cmd+Shift+Z: Redo
 * - Delete: Delete selected
 */
export function useEditorKeyboard({
  handlers,
  enabled = true
}: UseEditorKeyboardOptions) {
  const handleKeyDown = useCallback(
    (e: KeyboardEvent) => {
      if (!enabled) return;

      const isMac = /Mac|iPhone|iPad|iPod/.test(navigator.platform);
      const modKey = isMac ? e.metaKey : e.ctrlKey;

      // Ctrl/Cmd + S = Save
      if (modKey && e.key === "s") {
        e.preventDefault();
        handlers.onSave();
      }

      // Ctrl/Cmd + Z = Undo
      if (modKey && e.key === "z" && !e.shiftKey && handlers.onUndo) {
        e.preventDefault();
        handlers.onUndo();
      }

      // Ctrl/Cmd + Shift + Z = Redo
      if (modKey && e.key === "z" && e.shiftKey && handlers.onRedo) {
        e.preventDefault();
        handlers.onRedo();
      }

      // Delete = Delete selected
      if (e.key === "Delete" && handlers.onDelete) {
        handlers.onDelete();
      }
    },
    [enabled, handlers]
  );

  useEffect(() => {
    if (!enabled) return;

    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [enabled, handleKeyDown]);
}
```

```typescript
import { useState, useCallback, useRef } from "react";

export interface UseEditorCustomCssOptions {
  onCssChange?: (css: string) => void;
}

export interface CustomCssState {
  css: string;
  error: string | null;
  isValid: boolean;
}

/**
 * Hook to manage custom CSS editing and validation.
 *
 * Handles:
 * - Store CSS content
 * - Validate CSS syntax
 * - Inject into document
 * - Error reporting
 */
export function useEditorCustomCss({ onCssChange }: UseEditorCustomCssOptions = {}) {
  const [state, setState] = useState<CustomCssState>({
    css: "",
    error: null,
    isValid: true
  });

  // Ref to cache the <style> element
  const styleElementRef = useRef<HTMLStyleElement | null>(null);

  const updateCss = useCallback(
    (newCss: string) => {
      try {
        // Validate CSS by creating a StyleSheet
        // (This is a light validation; doesn't catch all errors)
        const styleEl = document.createElement("style");
        styleEl.textContent = newCss;
        document.head.appendChild(styleEl);

        // If we got here, CSS is valid
        document.head.removeChild(styleEl);

        setState({
          css: newCss,
          error: null,
          isValid: true
        });

        // Inject into document for real
        if (!styleElementRef.current) {
          styleElementRef.current = document.createElement("style");
          styleElementRef.current.id = "editor-custom-css";
          document.head.appendChild(styleElementRef.current);
        }
        styleElementRef.current.textContent = newCss;

        onCssChange?.(newCss);
      } catch (err) {
        const errorMsg = err instanceof Error ? err.message : "Invalid CSS";
        setState({
          css: newCss,
          error: errorMsg,
          isValid: false
        });
      }
    },
    [onCssChange]
  );

  return {
    ...state,
    updateCss
  };
}
```

```typescript
"use client";

import React, { createContext, ReactNode, useMemo } from 'react';
import { useEditorLock } from '@/hooks/editor';
import { useEditorAutosave } from '@/hooks/editor';
import { useEditorDocument } from '@/hooks/editor';
import { useEditorKeyboard } from '@/hooks/editor';
import { useEditorCustomCss } from '@/hooks/editor';
import { useEditorUi } from '@/hooks/editor';

export interface EditorContextValue {
  // Lock
  lock: ReturnType<typeof useEditorLock>;

  // Autosave
  autosave: ReturnType<typeof useEditorAutosave>;

  // Document
  document: ReturnType<typeof useEditorDocument>;

  // UI
  ui: ReturnType<typeof useEditorUi>;

  // Custom CSS
  customCss: ReturnType<typeof useEditorCustomCss>;

  // Actions
  actions: {
    save: () => void;
  };
}

export const EditorContext = createContext<EditorContextValue | null>(null);

export interface EditorProviderProps {
  landingId: string;
  userId: string;
  children: ReactNode;
}

/**
 * Provider component that aggregates all editor hooks and state.
 *
 * Wrap your editor components with this provider to access all state/actions.
 *
 * Usage:
 * <EditorProvider landingId="123" userId="user-456">
 *   <LandingEditorShell />
 * </EditorProvider>
 */
export function EditorProvider({
  landingId,
  userId,
  children,
}: EditorProviderProps) {
  // Setup all hooks
  const lock = useEditorLock({ landingId, userId });

  // Autosave needs serializeEditor function
  // (will be passed from child component after GrapesJS is initialized)
  const autosave = useEditorAutosave({
    landingId,
    serializeEditor: () => ({}), // TODO: wire up from GrapesJS
  });

  const document = useEditorDocument({ landingId });

  const ui = useEditorUi();

  const customCss = useEditorCustomCss();

  // Keyboard shortcuts
  useEditorKeyboard({
    handlers: {
      onSave: () => {
        console.log('Save shortcut triggered');
        // Trigger manual save
      },
    },
  });

  const value: EditorContextValue = useMemo(
    () => ({
      lock,
      autosave,
      document,
      ui,
      customCss,
      actions: {
        save: () => {
          // Manual save action
        },
      },
    }),
    [lock, autosave, document, ui, customCss]
  );

  return (
    <EditorContext.Provider value={value}>
      {children}
    </EditorContext.Provider>
  );
}

/**
 * Hook to consume editor context.
 *
 * Usage: const editor = useEditor();
 */
export function useEditor() {
  const context = React.useContext(EditorContext);
  if (!context) {
    throw new Error('useEditor must be used within EditorProvider');
  }
  return context;
}
```

## 📋 ШАГ 5: Wire up GrapesJS editor integration

### 5.1 Создай hook для GrapesJS инициализации

**Файл:** `apps/web/hooks/editor/use-grapes-editor.ts`

```typescript
import { useEffect, useRef } from "react";
import grapesjs from "grapesjs";
import { useEditor } from "@/context/editor-context";

/**
 * Initialize GrapesJS editor and wire up events.
 */
export function useGrapesEditor(containerId: string, landing: any) {
  const editor = useEditor();
  const grapeEditorRef = useRef<any>(null);

  useEffect(() => {
    if (!landing) return;

    // Initialize GrapesJS
    const grapesEditor = grapesjs.init({
      container: `#${containerId}`,
      components: landing.grapesJson?.components,
      style: landing.grapesJson?.styles
      // ... other config
    });

    grapeEditorRef.current = grapesEditor;

    // ============ WIRE UP DIRTY FLAG ============
    // On any change, mark as dirty so autosave will trigger
    grapesEditor.on("component:update", () => editor.autosave.markDirty());
    grapesEditor.on("style:update", () => editor.autosave.markDirty());
    grapesEditor.on("undo", () => editor.autosave.markDirty());
    grapesEditor.on("redo", () => editor.autosave.markDirty());

    return () => {
      grapesEditor.destroy();
    };
  }, [landing, containerId, editor.autosave]);

  return grapeEditorRef;
}
```
