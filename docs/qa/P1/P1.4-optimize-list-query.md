## 🎯 ПРОБЛЕМА

### Текущее состояние (危險 OVER-FETCHING):
```typescript
// apps/api/src/landings/landings.service.ts

async list(userId: string, skip: number, take: number) {
  return await this.prisma.landing.findMany({
    where: { ownerId: userId, deletedAt: null },
    include: {
      geo: true,              // ← Fetch all columns
      category: true,         // ← Fetch all columns
      variant: true,          // ← Fetch all columns
      template: true,         // ← Fetch all columns
      owner: { select: { ... } },
      currentVersion: true,   // ⚠️ INCLUDES grapesJson (5-50MB), html, css!
    },
    skip,
    take,
  });
}

// Problem:
// 50 landings × 5-50 MB grapesJson = 250-2500 MB over wire!
// Just to show a list of landing names
````

### Оптимальная approach

```typescript
// Only SELECT what's needed for list view
landing: {
  select: {
    id: true,
    name: true,
    slug: true,
    publicId: true,
    status: true,
    createdAt: true,
    updatedAt: true,
    geoId: true,
    categoryId: true,
    // NOT currentVersion (too big)
    // NOT owner.email (only show name)
    owner: { select: { id: true, name: true } },
    geo: { select: { id: true, name: true } },
    category: { select: { id: true, name: true } },
  }
}

// Result: 50 landings × ~2-5 KB = 100-250 KB
// 100x smaller payload!
```

  @Expose()

  @Expose()

  @Expose()

  @Expose()
  name: string;

 * Main landing list item DTO.
 *
 * This is what gets returned in list responses.

  @Expose()
  id: string;

  @Expose()
  name: string;

  @Expose()
  slug: string;

  @Expose()
  publicId: string;

  @Expose()

  @Expose()
  @Type(() => LandingListGeoDto)
  geo?: LandingListGeoDto;

  @Expose()
  @Type(() => LandingListCategoryDto)
  category?: LandingListCategoryDto;

  @Expose()

  variant?: LandingListVariantDto;

  @Expose()
  @Type(() => LandingListOwnerDto)
  owner?: LandingListOwnerDto;

  @Expose()
  createdAt: Date;

  @Expose()
  updatedAt: Date;

  @Expose()
  lastModifiedBy?: string;

  @Expose()
  thumbnailUrl?: string;

/**
 * Paginated list response.

export class LandingListResponseDto {
  @Expose()
  @Type(() => LandingListItemDto)
  items: LandingListItemDto[];

  @Expose()
  total: number;

  @Expose()
  skip: number;

  @Expose()
  take: number;

  @Expose()
  hasMore: boolean;
}

### 2.2 Обновить с тонким SELECT

**БЫЛО:**

```typescript
async list(userId: string, skip = 0, take = 10) {
  return await this.prisma.landing.findMany({
    where: { ownerId: userId, deletedAt: null },
    include: {
      geo: true,
      category: true,
      variant: true,
      template: true,
      owner: { select: { id: true, email: true, name: true } },
      currentVersion: true, // ⚠️ INCLUDES HUGE BLOBS
    },
    skip,
    take: Math.min(take, 100), // Cap at 100
    orderBy: { updatedAt: 'desc' },
  });
}
```

**СТАЛО:**

```typescript
import { LandingListItemDto, LandingListResponseDto } from './dto/landing-list.dto';
import { plainToInstance } from 'class-transformer';

async list(
  userId: string,
  skip = 0,
  take = 10
): Promise<LandingListResponseDto> {
  // ============ TIGHT SELECT ============
  // Only fetch what's needed for list view
  const [landings, total] = await Promise.all([
    this.prisma.landing.findMany({
      where: {
        ownerId: userId,
        deletedAt: null,
      },
      // ✅ Use select instead of include to reduce payload
      select: {
        id: true,
        name: true,
        slug: true,
        publicId: true,
        status: true,
        createdAt: true,
        updatedAt: true,

        // Relationships - only essential fields
        geoId: true,
        geo: {
          select: {
            id: true,
            name: true,
            // NOT: slug, code, region, etc
          },
        },

        categoryId: true,
        category: {
          select: {
            id: true,
            name: true,
          },
        },

        variantId: true,
        variant: {
          select: {
            id: true,
            name: true,
          },
        },

        ownerId: true,
        owner: {
          select: {
            id: true,
            email: true,
            name: true,
          },
        },

        // Metadata fields
        lockedById: true,
        lockExpires: true,

        // NOT: currentVersion (because it contains grapesJson)
        // NOT: versions (array of all versions)
        // NOT: audit logs
      },
      skip,
      take: Math.min(take, 100),
      orderBy: { updatedAt: 'desc' },
    }),

    // Parallel count query
    this.prisma.landing.count({
      where: {
        ownerId: userId,
        deletedAt: null,
      },
    }),
  ]);

  // ============ TRANSFORM TO DTO ============
  const items = plainToInstance(LandingListItemDto, landings, {
    excludeExtraneousValues: true,
  });

  const hasMore = skip + landings.length < total;

  return {
    items,
    total,
    skip,
    take,
    hasMore,
  };
}
```

## 📋 ШАГ 5: Добавить индексы для быстрого query

### 5.1 Обновить Prisma schema

**Файл:** `prisma/schema.prisma`

```prisma
model Landing {
  id        String      @id @default(cuid())
  name      String
  slug      String      @unique
  publicId  String      @unique
  status    LandingStatus @default(DRAFT)

  ownerId   String
  owner     User        @relation(fields: [ownerId], references: [id])

  geoId     String?
  geo       Geo?        @relation(fields: [geoId], references: [id])

  categoryId String?
  category   Category?   @relation(fields: [categoryId], references: [id])

  createdAt DateTime    @default(now())
  updatedAt DateTime    @updatedAt
  deletedAt DateTime?

  currentVersionId String?
  currentVersion   Version? @relation(name: "currentVersion", fields: [currentVersionId], references: [id])

  // ✅ ADD INDICES for list query
  @@index([ownerId])                    // For WHERE ownerId = ?
  @@index([ownerId, deletedAt])         // For WHERE ownerId AND deletedAt IS NULL
  @@index([status, deletedAt])
  @@unique([ownerId, slug])             // Unique per owner
}
```

## 🚨 ВОЗМОЖНЫЕ ОШИБКИ

| Ошибка                                            | Причина                           | Решение                                      |
| ------------------------------------------------- | --------------------------------- | -------------------------------------------- |
| "Cannot access property 'currentVersion' of null" | currentVersion не в select        | Это нормально - явно не включили в list view |
| Query timeout                                     | Index не все еще создан           | Запусти `npx prisma migrate dev`             |
| DTO свойства не в response                        | plainToInstance не трансформирует | Убедись что DTO имеет @Expose()              |
| "count is not a function"                         | Prisma версия старая              | Обновить `@prisma/client` до последней       |
